  ```ts
  export interface Turn { labels: readonly number[]; centers: readonly Pt[]; /** after the move */ changed: number | null; /** null on turn 1 */ inertia: number }
  export interface Run { id: string; init: 'k-means++' | 'random'; seed: number; start: readonly Pt[]; turns: readonly Turn[]; inertia: number }
  export interface NInitRun { seed: number; labels: readonly number[]; centers: readonly Pt[]; inertia: number; iterations: number }
  export interface KResult { k: number; labels: readonly number[]; centers: readonly Pt[]; inertia: number; sizes: readonly number[]; silhouette: number | null; s: readonly number[] | null; a: readonly number[] | null; b: readonly number[] | null }
  export interface KmeansData { meta: { sklearn: string; features: readonly [string, string]; n: number }; points: readonly Pt[]; runs: readonly Run[]; nInit: readonly NInitRun[]; nInitBest: number; ks: readonly KResult[]; distancePoints: readonly number[]; silhouettePoint: number }
  export const kmeansData: KmeansData;
  export function kResult(k: number): KResult; // 1..10, throws otherwise
  ```
  Kurallar: `runs[0]` iyi başlangıç, `runs[1]` kötü başlangıç, toplam 12 run (spec "Başlangıçlar…" bölümü); `Turn` t: atama `labels` (t'den önceki merkezlere göre), sonra `centers` = atanmış kümelerin ortalaması, `inertia` = `labels`'ın yeni `centers`'a kare uzaklık toplamı; son turun `changed`'i 0. `ks[i].k === i + 1`. `distancePoints` 3 indeks (runs[0] tur 1), `silhouettePoint` 1 indeks (k=3) — kuralları spec'te.
